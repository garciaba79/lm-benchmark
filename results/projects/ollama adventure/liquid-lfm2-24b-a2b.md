```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Ollama's Adventure</title>
    <style>
        body {
            margin: 0;
            overflow: hidden;
            background: linear-gradient(to bottom, #87CEEB, #FFD700);
            display: flex;
            justify-content: center;
            align-items: center;
            height: 100vh;
            font-family: 'Comic Sans MS', cursive, sans-serif;
        }
        canvas {
            border: 2px solid #FFD700;
            background: #000;
            display: block;
        }
        #game-over {
            position: absolute;
            top: 50%;
            left: 50%;
            transform: translate(-50%, -50%);
            display: none;
            text-align: center;
            color: white;
            font-size: 24px;
            background: rgba(0, 0, 0, 0.7);
            padding: 20px;
            border-radius: 10px;
        }
        #retry-btn {
            margin-top: 20px;
            padding: 10px 20px;
            font-size: 18px;
            cursor: pointer;
            background: #FFD700;
            color: black;
            border: none;
            border-radius: 5px;
        }
    </style>
</head>
<body>
    <canvas id="gameCanvas" width="800" height="400"></canvas>
    <div id="game-over">
        <p>Game Over!</p>
        <p>Score: <span id="score">0</span></p>
        <button id="retry-btn">Retry</button>
    </div>
    <script>
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const gameOverDiv = document.getElementById('game-over');
        const scoreDisplay = document.getElementById('score');
        const retryBtn = document.getElementById('retry-btn');

        let alpaca = {
            x: 50,
            y: canvas.height / 2 - 30,
            width: 40,
            height: 40,
            speed: 2,
            jumpHeight: 0,
            gravity: 0.5,
            velocityY: 0,
            onGround: false
        };

        let obstacles = [];
        let score = 0;
        let gameSpeed = 1;
        let gameOver = false;
        let audioContext;
        let jumpSound, scoreSound, hitSound;

        function initAudio() {
            audioContext = new (window.AudioContext || window.webkitAudioContext)();
            const oscillator = audioContext.createOscillator();
            const gainNode = audioContext.createGain();

            // Jump sound
            oscillator.type = 'square';
            oscillator.frequency.setValueAtTime(800, audioContext.currentTime);
            oscillator.connect(gainNode);
            gainNode.gain.setValueAtTime(0.1, audioContext.currentTime);
            jumpSound = oscillator;

            // Score sound
            oscillator.type = 'sine';
            oscillator.frequency.setValueAtTime(1000, audioContext.currentTime);
            oscillator.connect(gainNode);
            gainNode.gain.setValueAtTime(0.1, audioContext.currentTime);
            scoreSound = oscillator;

            // Hit sound
            oscillator.type = 'triangle';
            oscillator.frequency.setValueAtTime(500, audioContext.currentTime);
            oscillator.connect(gainNode);
            gainNode.gain.setValueAtTime(0.1, audioContext.currentTime);
            hitSound = oscillator;
        }

        function drawAlpaca() {
            ctx.fillStyle = '#FFD700';
            ctx.fillRect(alpaca.x, alpaca.y, alpaca.width, alpaca.height);
            ctx.strokeStyle = '#000';
            ctx.strokeRect(alpaca.x, alpaca.y, alpaca.width, alpaca.height);

            // Draw face
            ctx.beginPath();
            ctx.arc(alpaca.x + 20, alpaca.y + 15, 10, 0, Math.PI * 2);
            ctx.fillStyle = '#000';
            ctx.fill();
            ctx.strokeStyle = '#000';
            ctx.stroke();
        }

        function drawObstacles() {
            obstacles.forEach((obstacle, index) => {
                ctx.fillStyle = '#FF0000';
                ctx.fillRect(obstacle.x, obstacle.y, obstacle.width, obstacle.height);
                ctx.strokeStyle = '#000';
                ctx.strokeRect(obstacle.x, obstacle.y, obstacle.width, obstacle.height);

                // Draw shadow
                ctx.fillStyle = 'rgba(255, 0, 0, 0.5)';
                ctx.fillRect(obstacle.x, obstacle.y + 10, obstacle.width, obstacle.height - 10);
            });
        }

        function updateAlpaca() {
            if (gameOver) return;

            // Gravity
            alpaca.velocityY += alpaca.gravity;
            alpaca.y += alpaca.velocityY;

            // Jump
            if (alpaca.onGround && (keyPressed('Space') || keyPressed('Enter'))) {
                alpaca.velocityY = -alpaca.jumpHeight;
                alpaca.onGround = false;
                jumpSound.play();
            }

            // Move obstacles
            obstacles.forEach(obstacle => {
                obstacle.x -= obstacle.speed * gameSpeed;
                if (obstacle.x + obstacle.width < 0) {
                    score += 10; // Score for each obstacle avoided
                    scoreDisplay.textContent = score;
                    scoreSound.play();
                    obstacles = obstacles.filter(o => o.x + o.width > 0);
                }
            });

            // Check collision with ground
            if (alpaca.y + alpaca.height >= canvas.height) {
                gameOver = true;
                gameOverDiv.style.display = 'block';
                document.getElementById('jump-btn').style.display = 'none';
            }

            // Check collision with obstacles
            obstacles.forEach(obstacle => {
                if (alpaca.x < obstacle.x + obstacle.width &&
                    alpaca.x + alpaca.width > obstacle.x &&
                    alpaca.y < obstacle.y + obstacle.height &&
                    alpaca.y + alpaca.height > obstacle.y) {
                    gameOver = true;
                    gameOverDiv.style.display = 'block';
                    document.getElementById('jump-btn').style.display = 'block';
                }
            });

            // Increase speed over time
            if (score % 50 === 0 && score !== 0) {
                gameSpeed += 0.1;
            }
        }

        function drawBackground() {
            // Parallax background layers
            ctx.fillStyle = 'url(#gradient1)';
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            const gradient1 = ctx.createLinearGradient(0, 0, canvas.width, canvas.height);
            gradient1.addColorStop(0, '#87CEEB');
            gradient1.addColorStop(1, '#FFD700');
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            const gradient2 = ctx.createLinearGradient(0, 0, canvas.width, canvas.height);
            gradient2.addColorStop(0, '#FFD700');
            gradient2.addColorStop(1, '#FF4500');
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            // Draw moving background elements
            const bg1 = ctx.createPattern(gradient1, 'repeat');
            const bg2 = ctx.createPattern(gradient2, 'repeat');
            ctx.fillStyle = bg1;
            ctx.fillRect(-canvas.width / 2, -canvas.height / 2, canvas.width * 2, canvas.height);
            ctx.fillStyle = bg2;
            ctx.fillRect(-canvas.width / 2, -canvas.height / 2, canvas.width * 2, canvas.height);

            // Animate background
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            drawBackground();
        }

        function gameLoop() {
            if (gameOver) return;

            ctx.clearRect(0, 0, canvas.width, canvas.height);
            drawBackground();

            // Update and draw alpaca
            updateAlpaca();
            drawAlpaca();

            // Draw obstacles
            updateObstacles();
            drawObstacles();

            // Draw score
            ctx.fillStyle = '#000';
            ctx.font = '20px Arial';
            ctx.fillText('Score: ' + score, 10, 30);

            requestAnimationFrame(gameLoop);
        }

        function updateObstacles() {
            const obstacleWidth = Math.random() * 50 + 30;
            const obstacleHeight = 40;
            const obstacleX = canvas.width + Math.random() * (canvas.width - obstacleWidth);
            obstacles.push({ x: obstacleX, y: canvas.height - obstacleHeight, width: obstacleWidth, height: obstacleHeight, speed: 2 + Math.random() * 1.5 });
        }

        function keyPressed(key) {
            return key === ' ' || key === 'Enter';
        }

        retryBtn.addEventListener('click', () => {
            // Reset game
            alpaca = {
                x: 50,
                y: canvas.height / 2 - 30,
                width: 40,
                height: 40,
                speed: 2,
                jumpHeight: 0,
                gravity: 0.5,
                velocityY: 0,
                onGround: false
            };
            obstacles = [];
            score = 0;
            gameSpeed = 1;
            gameOver = false;
            scoreDisplay.textContent = score;
            audioContext = new (window.AudioContext || window.webkitAudioContext)();
            initAudio();
            gameLoop();
        });

        // Start the game
        initAudio();
        gameLoop();
    </script>
</body>
</html>
```
